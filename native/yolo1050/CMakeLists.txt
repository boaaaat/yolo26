cmake_minimum_required(VERSION 3.24)
project(yolo1050 LANGUAGES CXX CUDA)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "yolo1050 requires Windows x64 and MSVC")
endif()
if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 11.8 OR NOT CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.0)
  message(FATAL_ERROR "Select CUDA 11.8 explicitly; the GTX 1050 deployment uses the TensorRT 8.6.1 CUDA 11.8 SDK")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 61)
set(TENSORRT_ROOT "" CACHE PATH "Extracted TensorRT 8.6.1 Windows CUDA 11.8 SDK")
find_package(CUDAToolkit 11.8 REQUIRED)
if(CUDAToolkit_VERSION VERSION_LESS 11.8 OR NOT CUDAToolkit_VERSION VERSION_LESS 12.0)
  message(FATAL_ERROR "CUDA runtime libraries must also come from the CUDA 11.8 toolkit")
endif()
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
find_path(TRT_INCLUDE NvInfer.h PATHS "${TENSORRT_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
file(READ "${TRT_INCLUDE}/NvInferVersion.h" TRT_VERSION_HEADER)
if(NOT TRT_VERSION_HEADER MATCHES "NV_TENSORRT_MAJOR +8" OR
   NOT TRT_VERSION_HEADER MATCHES "NV_TENSORRT_MINOR +6" OR
   NOT TRT_VERSION_HEADER MATCHES "NV_TENSORRT_PATCH +1")
  message(FATAL_ERROR "TensorRT 8.6.1 headers required; do not use the Blackwell SDK")
endif()
foreach(lib nvinfer nvonnxparser nvinfer_plugin)
  find_library(${lib}_LIBRARY ${lib} PATHS "${TENSORRT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
endforeach()
find_package(nlohmann_json 3.11.3 QUIET)
if(NOT nlohmann_json_FOUND)
  include(FetchContent)
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(json)
endif()
add_executable(yolo1050 src/main.cpp src/engine.cpp src/capture.cpp src/controls.cpp src/preprocess.cu)
target_include_directories(yolo1050 PRIVATE include "${TRT_INCLUDE}" ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(yolo1050 PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(yolo1050 PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4>
  $<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CONFIG:Release>>:/O2>)
target_link_libraries(yolo1050 PRIVATE CUDA::cudart ${nvinfer_LIBRARY} ${nvonnxparser_LIBRARY}
  ${nvinfer_plugin_LIBRARY} ${OpenCV_LIBS} nlohmann_json::nlohmann_json
  d3d11 dxgi user32 gdi32 winmm bcrypt)
install(TARGETS yolo1050 RUNTIME DESTINATION .)
